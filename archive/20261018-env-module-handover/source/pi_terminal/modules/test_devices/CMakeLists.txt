foreach(kind environment counter camera)
    if(kind STREQUAL "environment")
        set(ID 129)
        set(TITLE "环境监测（测试）")
        set(ACTIONS "\"open\",\"start\",\"stop\",\"reset\",\"request_disconnect\"")
    elseif(kind STREQUAL "counter")
        set(ID 130)
        set(TITLE "计数器（测试）")
        set(ACTIONS "\"open\",\"increment\",\"decrement\",\"reset\",\"request_disconnect\"")
    else()
        set(ID 131)
        set(TITLE "相机（测试）")
        set(ACTIONS "\"open\",\"start\",\"stop\",\"frame\",\"request_disconnect\"")
    endif()
    set(MANIFEST_CONTENT "{\"id\":\"test.${kind}\",\"name\":\"${TITLE}\",\"version\":\"1.0.0\",\"module_id\":${ID},\"package_id\":\"test.${kind}-1.0.0-${CMAKE_SYSTEM_PROCESSOR}\",\"package_version\":\"1.0.0\",\"host_arch\":[\"${CMAKE_SYSTEM_PROCESSOR}\"],\"library\":\"lib${kind}_module.so\",\"qml_page\":\"qml/ExamplePage.qml\",\"control_protocol\":\"pogo-control-v1\",\"data_protocol\":\"module-defined-v1\",\"abi_version\":1,\"min_host_api\":1,\"max_payload\":65532,\"qt_major\":6,\"unload_contract\":\"synchronous-stop-v1\",\"capabilities\":[\"sample.read\",\"sample.configure\"],\"permissions\":[\"transport.hotspot_lan\"],\"actions\":[${ACTIONS}]}")
    file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/${kind}/qml")
    file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/${kind}/manifest.json" "${MANIFEST_CONTENT}")
    configure_file(../example/example_manifest.h.in ${kind}/example_manifest.h @ONLY)
    configure_file(${kind}.qml ${kind}/qml/ExamplePage.qml COPYONLY)
    add_library(${kind}_module SHARED ../example/example_module.cpp)
    target_include_directories(${kind}_module PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/${kind}")
    target_link_libraries(${kind}_module PRIVATE OrangePi::ModuleSDK)
    set_target_properties(${kind}_module PROPERTIES LIBRARY_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/${kind}")
    install(TARGETS ${kind}_module LIBRARY DESTINATION lib/orange-pi-terminal/modules/${ID})
    install(FILES "${CMAKE_CURRENT_BINARY_DIR}/${kind}/manifest.json" DESTINATION lib/orange-pi-terminal/modules/${ID})
    install(DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/${kind}/qml" DESTINATION lib/orange-pi-terminal/modules/${ID})
endforeach()
