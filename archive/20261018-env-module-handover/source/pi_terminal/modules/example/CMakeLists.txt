file(GLOB EXAMPLE_QML_SOURCES CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/qml/*.qml")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS
    "${CMAKE_CURRENT_SOURCE_DIR}/manifest.json" ${EXAMPLE_QML_SOURCES})
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/manifest.json" MANIFEST_CONTENT)
string(REPLACE "aarch64" "${CMAKE_SYSTEM_PROCESSOR}" MANIFEST_CONTENT "${MANIFEST_CONTENT}")
string(REPLACE "-arm64" "-${CMAKE_SYSTEM_PROCESSOR}" MANIFEST_CONTENT "${MANIFEST_CONTENT}")
string(JSON EXAMPLE_MODULE_ID ERROR_VARIABLE MANIFEST_ID_ERROR GET "${MANIFEST_CONTENT}" module_id)
if(MANIFEST_ID_ERROR)
    message(FATAL_ERROR "example manifest.json must contain a numeric module_id: ${MANIFEST_ID_ERROR}")
endif()
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/manifest.json" "${MANIFEST_CONTENT}")
configure_file(example_manifest.h.in example_manifest.h @ONLY)
add_library(example_module SHARED example_module.cpp)
target_include_directories(example_module PRIVATE ${CMAKE_CURRENT_BINARY_DIR})
target_link_libraries(example_module PRIVATE OrangePi::ModuleSDK)
target_compile_features(example_module PRIVATE cxx_std_17)
set_target_properties(example_module PROPERTIES OUTPUT_NAME example_module)
file(COPY qml DESTINATION ${CMAKE_CURRENT_BINARY_DIR})
set(EXAMPLE_INSTALL_DIR "lib/orange-pi-terminal/modules/${EXAMPLE_MODULE_ID}")
install(TARGETS example_module LIBRARY DESTINATION "${EXAMPLE_INSTALL_DIR}")
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/manifest.json" DESTINATION "${EXAMPLE_INSTALL_DIR}")
install(DIRECTORY qml DESTINATION "${EXAMPLE_INSTALL_DIR}")
