cmake_minimum_required(VERSION 3.16)
project(lcd_view_regression C)
set(CMAKE_C_STANDARD 99)
if(EXISTS "${CMAKE_CURRENT_LIST_DIR}/../../lvgl/src")
    set(DEFAULT_LVGL_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../lvgl")
else()
    set(DEFAULT_LVGL_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../../../shared/lvgl")
endif()
set(LVGL_ROOT "${DEFAULT_LVGL_ROOT}" CACHE PATH "LVGL runtime source root")
file(GLOB_RECURSE LVGL_SOURCES "${LVGL_ROOT}/src/*.c")
add_library(lvgl STATIC ${LVGL_SOURCES})
target_include_directories(lvgl PUBLIC "${LVGL_ROOT}" "${LVGL_ROOT}/src")
target_compile_definitions(lvgl PUBLIC LV_CONF_SKIP=1 LV_COLOR_DEPTH=16 LV_COLOR_16_SWAP=0 LV_LVGL_H_INCLUDE_SIMPLE=1)
file(GLOB FONTS "${CMAKE_CURRENT_LIST_DIR}/../fonts/*.c")
add_executable(lcd_view_test lcd_view_test.c "../ui_env_view.c" ${FONTS})
target_include_directories(lcd_view_test PRIVATE "../include" "../../lcd_st7789/include")
target_link_libraries(lcd_view_test PRIVATE lvgl m)
enable_testing()
add_test(NAME lcd_pixels_layout_and_wire_order COMMAND lcd_view_test)
